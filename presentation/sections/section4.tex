\section{مدل}

\begin{frame}{چه ساخته شد و مدل به چه چیزی نگاه می‌کند}
  \begin{itemize}
    \item ۱۰۰ ویژگی، ۱۳ خانواده‌ی مدل، غربال چهارمرحله‌ای، انتخاب نهایی یک مدل درختی
      کوانتایلی
    \item پروتکل: تقسیم زمانی با مبدأ غلتان؛ پنجره‌ی آزمون فقط یک‌بار باز شد
    \item مهم‌ترین ورودی‌ها: هویت سلف، برهم‌کنش روزهفته با شهر، حجم رزرو، نرخ دیروزِ همان
      سلول، شوک دیروز
  \end{itemize}

\begin{frame}{چطور مدل‌ها ارزیابی شدند: دو نوع پنجره}
  \begin{itemize}
    \item ۵ دور اعتبارسنجی با مبدأ آموزش ثابت؛ فقط انتهای آموزش و محل آزمون جلوتر می‌رود
    \item همین ۵ دور مدل برتر را انتخاب کردند، نه پنجره‌ی انتهایی
    \item پنجره‌ی انتهایی قفل، فقط یک‌بار و بعد از انتخاب مدل، به‌عنوان آزمون فشار باز شد
  \end{itemize}
  \begin{figure}
    \centering
    \includegraphics[width=0.64\textwidth]{slides_14a_eval_windows}
  \end{figure}
  \note{اگر استاد پرسید «چرا پنجره‌ی لغزان نه»: با ۱۴۲ روز سرو دورریختن داده‌ی قدیمی
  قابل‌توجیه نبود. پنجره‌ی انتهایی عمداً به‌عنوان آزمون فشار طبقه‌بندی شد چون شامل بازگشت
  پس از شکاف ۲۹روزه‌ی رمضان و سه روز سوگواری ملی است — یک رژیم غیرعادی که رتبه‌بندی مدل‌ها
  را گمراه می‌کند، پس نباید معیار انتخاب باشد. شاهد: \lr{src/cv.py}
  (\lr{WalkForwardSplitter}، \lr{holdout\_split})، \lr{data/processed/cv\_folds.json}،
  \lr{doc/data\_manifest.md} (دروازه‌ی \lr{cv\_folds\_hash})، ردیف ۴۹ decision\_log.}
\end{frame}

\begin{frame}{۱۳ خانواده‌ی مدل امتحان شدند}
  \begin{itemize}
    \item از رگرسیون خطی ساده تا شبکه‌ی عصبی و مدل‌های بیزی — طیف کامل، نه فقط چند حدس
    \item هر خانواده حداقل یک نماینده تا انتهای مسیر آموزش و ارزیابی رفت
    \item خانواده‌ی درختی/بوستینگ (خ۲) در نهایت مدل برتر را داد؛ ترکیب و کالیبراسیون (خ۱۲،
      خ۱۳) لایه‌ای روی همه‌اند، نه یک رقیب مستقل
  \end{itemize}

  یا پیچیده‌ترین مدل کافی است — نتیجه‌ی بخش بعد (نتایج منفی) دقیقاً همین را نشان می‌دهد. شاهد:

\begin{frame}{مدل برتر چطور کار می‌کند}
  \begin{itemize}
    \item یک \lr{LightGBM} کوانتایلی: صدها درخت تصمیم پشت‌سرهم، هرکدام فقط خطای باقی‌مانده‌ی
      درخت‌های قبلی را اصلاح می‌کند
    \item خروجی مستقیماً همان چندک $\tau=0.20$ است که برای مسئله انتخاب شده بود، نه یک
      میانگین که بعداً تبدیل شود
    \item روی \lr{fold}های رسمی از قوی‌ترین خط پایه (\lr{B3}) به‌طور معنادار برد
  \end{itemize}
