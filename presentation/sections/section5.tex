\begin{frame}{چند درصد از هدررفت جلوگیری می‌شود}
  \begin{itemize}
    \item \textbf{$71\%$ هدررفت حذف می‌شود، $29\%$ می‌ماند} — باقی‌مانده حدود ۶۷ هزار پرس در
      سال، یعنی $25.2$ میلیارد تومان
    \item برحسب نرخ عدم‌دریافت: از $8.05\%$ رزروها به $2.33\%$ می‌رسد
    \item سقف تئوریک (پیش‌بینی کامل و بی‌خطا) $100\%$ است؛ فاصله تا آن سقف عمداً برای
      اجتناب از کمبود گذاشته شده، نه ضعف مدل
  \end{itemize}
  \begin{figure}
    \centering
    \includegraphics[width=0.52\textwidth]{slides_18_waste_rate}
  \end{figure}
  \note{این اسلاید جلوی سوءتفاهم رایج را می‌گیرد. مدل هدررفت را صفر نمی‌کند و نباید بکند؛
  $29\%$ باقی‌مانده بهای بیمه‌ی نخوردن کمبود است.}
\end{frame}

\begin{frame}{طرف دیگر معامله: کمبود}
  \begin{itemize}
    \item $15.4\%$ سلول‌ها کمبود می‌خورند، ولی \textbf{عمق میانگین هر کمبود $4.2$ پرس} است
    \item در یک وعده در کل دانشگاه: میانگین ۱۸ پرس، میانه ۸ پرس؛ سالانه حدود $0.28\%$
      تقاضا (سطح خدمت $99.72\%$)
    \item به ازای هر ۱ پرس کمبود، حدود \textbf{۲۴ پرس} هدررفت جلوگیری می‌شود
  \end{itemize}
  \begin{figure}
    \centering
    \includegraphics[width=0.85\textwidth]{slides_19_tradeoff_bars}
  \end{figure}
  \note{اگر استاد عدد خالص خواست: با فرض هزینه‌ی کمبود چهار برابر مازاد، هزینه‌ی کمبود
  سالانه حدود $10.4$ میلیارد و سود خالص حدود $51.5$ میلیارد تومان می‌شود — این عدد در مخزن
  نیست، محاسبه‌ی مشتق است؛ اگر آوردی فرضش را هم بگو. شاهد:
  \lr{reports/phase10/10.3\_policy\_scenarios.md}، \lr{10.4\_annual\_savings.md}.}
\end{frame}
