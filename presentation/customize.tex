% TYPESETTING ELEMENTS

% شکل نمایش هر بخش در فهرست مطالب
\setbeamertemplate{section in toc}{$\blacktriangleright$~\inserttocsection}

% شکل نمایش هر زیربخش در فهرست مطالب — خالی (فقط سطح بخش نمایش داده می‌شود)
\setbeamertemplate{subsection in toc}{}

% زیرعنوان خودکار هر اسلاید: شماره + نام بخش جاری
\makeatletter
    \pretocmd\beamer@checkframetitle{\framesubtitle{\thesection \, \secname}}
\makeatother

% از شماره‌گذاری دوباره‌ی اسلایدهای شکسته‌شده به چند صفحه جلوگیری می‌کند
\setbeamertemplate{frametitle continuation}{}

% ابتدای هر بخش: یک اسلاید فهرست مطالب با بخش جاری برجسته — لوگوی گوشه هم موقتاً
% خاموش می‌شود چون روی پس‌زمینه‌ی آبی رنگش با آرم یکی می‌شود (هر دو آبی‌فیروزه‌ای دانشگاه)
\AtBeginSection[]
{
    \begingroup
    \setbeamertemplate{footline}{}
    \setbeamertemplate{headline}{}
    \themecolor{blue}
    \begin{frame}{\strtoc}
        \tableofcontents[currentsection]
    \end{frame}
    \endgroup
}

% ابتدای هر زیربخش (در این ارائه استفاده نشده، ولی برای تکمیل قالب نگه داشته شد)
\AtBeginSubsection[]
{
    \begin{frame}{\,}{\thesection \, \secname}
        \centering\textsl{\textbf{\textcolor{sintefblue}{
            \large \strsubsec
            \vskip15pt
            \LARGE \subsecname
        }}}
    \end{frame}
}

% اسلاید تک‌جمله‌ای تمام‌صفحه (برای یک ادعای محوری/گذار بین بخش‌ها)
\newcommand{\aboutpage}[2]{
    \begingroup
    \setbeamertemplate{footline}{}
    \setbeamertemplate{headline}{}
    \themecolor{blue}
    \begin{frame}[c]{#1}{\,}
        \centering
        \begin{minipage}{\textwidth}
            \usebeamercolor[fg]{normal text}
            \centering
            \Large \textsl{\normalsize #2}
        \end{minipage}
    \end{frame}
    \endgroup
}

% اسلاید مراجع (در این ارائه استفاده نشده، برای تکمیل قالب نگه داشته شد)
\newcommand{\bibliographpage}{
    \section{مراجع}
    \begingroup
    \themecolor{blue}
    \begin{frame}[allowframebreaks]{مراجع}{\,}
        \tiny
        \printbibliography[heading=none]
    \end{frame}
\endgroup
}
